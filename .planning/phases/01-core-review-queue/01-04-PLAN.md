3. In index.tsx, track draft index and implement advance logic:
```tsx
const [selectedIndex, setSelectedIndex] = useState<number>(-1)

const handleRowClick = (draft: Draft, index: number) => {
  setSelectedDraft(draft)
  setSelectedIndex(index)
  setSheetOpen(true)
}

const handleActionComplete = (action: 'approve' | 'reject') => {
  // Only auto-advance when viewing pending tab
  if (activeTab !== 'pending') {
    setSheetOpen(false)
    return
  }

  // The current draft will be removed from the list after the action
  // So the "next" draft will be at the same index
  // But we need to wait for the query to refetch

  // Small delay to let Convex update propagate
  setTimeout(() => {
    // At this point, drafts should be updated without the actioned draft
    // The draft at selectedIndex is now the "next" one
    if (drafts && selectedIndex < drafts.length) {
      setSelectedDraft(drafts[selectedIndex])
      // Keep sheet open with new draft
    } else if (drafts && drafts.length > 0) {
      // If we were at the end, show the last remaining draft
      setSelectedDraft(drafts[drafts.length - 1])
      setSelectedIndex(drafts.length - 1)
    } else {
      // No more pending drafts
      setSheetOpen(false)
      toast.success('Todos os drafts pendentes foram revisados!')
    }
  }, 500) // Wait for Convex real-time update
}

// Update table row onClick:
{drafts.map((draft, index) => (
  <TableRow
    key={draft._id}
    className="cursor-pointer hover:bg-muted/50"
    onClick={() => handleRowClick(draft, index)}
  >
    ...
  </TableRow>
))}
```

4. Pass the callback to DraftDetailSheet:
```tsx
<DraftDetailSheet
  draft={selectedDraft}
  open={sheetOpen}
  onOpenChange={handleSheetClose}
  onActionComplete={handleActionComplete}
/>
```

Note: The real-time nature of Convex means the drafts array updates automatically when a draft is approved/rejected. The setTimeout gives time for this update to propagate.
